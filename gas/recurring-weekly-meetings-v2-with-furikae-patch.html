// 校時セルの表記（例: "A火5" "B月3"）を { weekday: '火', period: 5 } に変換する。 // 表記に一致しなければ null を返す（"授"や"－"など通常の値の場合） function parsePeriodLabel_(v) { if (v === null || v === undefined) return null; const s = String(v).trim(); const m = s.match(/^[AB]?([月火水木金土日])([0-9０-９]+)$/); if (!m) return null; const periodStr = m[2].replace(/[０-９]/g, function(ch) { return String.fromCharCode(ch.charCodeAt(0) - 0xFEE0); }); const period = parseInt(periodStr, 10); if (!period) return null; return { weekday: m[1], period: period }; } // 行事取込の1行（32列）の中に、指定の曜日・時限の校時が実際に // 組まれているか調べる（1〜3年のいずれかの学年で一致すればtrue）。 // 振替で表記が変わっていれば、そちらに従って判定される。 function rowHasMeetingAtPeriod_(row, weekday, period) { if (!row) return false; const periodCols = [ EVENT_COL.G1_PERIODS[period - 1], EVENT_COL.G2_PERIODS[period - 1], EVENT_COL.G3_PERIODS[period - 1] ]; for (let i = 0; i < periodCols.length; i++) { const colIdx = periodCols[i]; if (!colIdx) continue; const label = parsePeriodLabel_(row[colIdx - 1]); if (label && label.weekday === weekday && label.period === period) return true; } return false; } // 曜日・時限固定の定例会議を本日・明日の予定に追加 // 行事取込の校時表記（振替も含む）を見て判定し、 // 行事マスター由来の予定と同じ上書き/非表示の仕組み（overrides）をそのまま使う try { const todayRow = findEventRow_(ss, date); const tomorrowRow = findEventRow_(ss, nextDate); RECURRING_WEEKLY_MEETINGS.forEach(meeting => { try { const contexts = [ { row: todayRow, target: today, targetDate: date }, { row: tomorrowRow, target: tomorrow, targetDate: nextDate } ]; contexts.forEach(ctx => { if (!rowHasMeetingAtPeriod_(ctx.row, meeting.weekday, meeting.period)) return; const name = meeting.name; const keyDateStr = Utilities.formatDate(ctx.targetDate, 'JST', 'yyyy-MM-dd'); const newKey = keyDateStr + '||' + name; const oldKubun = (ctx.target === today) ? 'today' : 'tomorrow'; const oldKey = keyDateStr + '|' + oldKubun + '|' + name; const override = overrides[newKey] || overrides[oldKey]; if (override && override.hide) return; ctx.target.push({ time: override ? (override.time || '') : '', name: override ? override.name : name, source: 'master', origName: name, isOverridden: !!override, overrideId: override ? override.id : null, overriddenBy: override ? (override.author || '') : '', overriddenAt: override ? (override.at || '') : '' }); }); } catch (e) { console.log('定例会議「' + meeting.name + '」の処理でエラー: ' + (e.message || e)); } }); } catch (e) { console.log('定例会議の処理全体でエラー: ' + (e.message || e)); } let weeklyEventText = row ? String(row[EVENT_COL.EVENT - 1] || '').trim() : ''; const weeklyRecurringNames = RECURRING_WEEKLY_MEETINGS .filter(m => rowHasMeetingAtPeriod_(row, m.weekday, m.period)) .map(m => m.name); if (weeklyRecurringNames.length) { weeklyEventText = weeklyEventText ? weeklyEventText + '・' + weeklyRecurringNames.join('・') : weeklyRecurringNames.join('・'); }
