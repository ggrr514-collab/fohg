// 曜日・時限固定の定例会議（行事取込には無いもの）。 // 本日・明日の予定・週報に自動表示する。行事取込の学年別校時列にある // 「A火5」のような表記（週+曜日+時限）を見て、実際にその時限が // 開かれている校時番号を探し、会議名の末尾（○h）はその実際の番号にする // （振替で月曜3限が火曜5限の内容になっていれば「研究部会3h」と表示）。 // 会議を追加・変更したい場合は、この配列を直接書き換えてください。 // weekday/period は「本来（振替が無い場合）の曜日・時限」を指定する const RECURRING_WEEKLY_MEETINGS = [ { weekday: '月', period: 3, baseName: '運営委員会' }, { weekday: '火', period: 5, baseName: '研究部会' }, { weekday: '水', period: 3, baseName: '教育相談部会' }, { weekday: '金', period: 3, baseName: '生徒指導委員会' } ]; // 行事取込の1行（32列）の中から、指定の曜日・時限（会議の本来の位置）の // ラベルが「実際に」何時限目にあるかを探す。振替が無ければ通常は // その曜日・時限と同じ番号が返る。見つからなければ null。 // 1〜3年のいずれかの学年で最初に見つかったものを採用する。 function findMeetingActualPeriod_(row, weekday, period) { if (!row) return null; const gradeColSets = [EVENT_COL.G1_PERIODS, EVENT_COL.G2_PERIODS, EVENT_COL.G3_PERIODS]; for (let p = 1; p <= 6; p++) { for (let g = 0; g < gradeColSets.length; g++) { const colIdx = gradeColSets[g][p - 1]; if (!colIdx) continue; const label = parsePeriodLabel_(row[colIdx - 1]); if (label && label.weekday === weekday && label.period === period) { return p; } } } return null; } // 曜日・時限固定の定例会議を本日・明日の予定に追加 // 行事取込の校時表記から「実際に開かれる時限」を探し、その番号で // 会議名（○h）を組み立てる。行事マスター由来の予定と同じ // 上書き/非表示の仕組み（overrides）をそのまま使う try { const todayRow = findEventRow_(ss, date); const tomorrowRow = findEventRow_(ss, nextDate); RECURRING_WEEKLY_MEETINGS.forEach(meeting => { try { const contexts = [ { row: todayRow, target: today, targetDate: date }, { row: tomorrowRow, target: tomorrow, targetDate: nextDate } ]; contexts.forEach(ctx => { const actualPeriod = findMeetingActualPeriod_(ctx.row, meeting.weekday, meeting.period); if (!actualPeriod) return; const name = meeting.baseName + actualPeriod + 'h'; const keyDateStr = Utilities.formatDate(ctx.targetDate, 'JST', 'yyyy-MM-dd'); const newKey = keyDateStr + '||' + name; const oldKubun = (ctx.target === today) ? 'today' : 'tomorrow'; const oldKey = keyDateStr + '|' + oldKubun + '|' + name; const override = overrides[newKey] || overrides[oldKey]; if (override && override.hide) return; ctx.target.push({ time: override ? (override.time || '') : '', name: override ? override.name : name, source: 'master', origName: name, isOverridden: !!override, overrideId: override ? override.id : null, overriddenBy: override ? (override.author || '') : '', overriddenAt: override ? (override.at || '') : '' }); }); } catch (e) { console.log('定例会議「' + meeting.baseName + '」の処理でエラー: ' + (e.message || e)); } }); } catch (e) { console.log('定例会議の処理全体でエラー: ' + (e.message || e)); } let weeklyEventText = row ? String(row[EVENT_COL.EVENT - 1] || '').trim() : ''; const weeklyRecurringNames = RECURRING_WEEKLY_MEETINGS .map(m => { const actualPeriod = findMeetingActualPeriod_(row, m.weekday, m.period); return actualPeriod ? (m.baseName + actualPeriod + 'h') : null; }) .filter(Boolean); if (weeklyRecurringNames.length) { weeklyEventText = weeklyEventText ? weeklyEventText + '・' + weeklyRecurringNames.join('・') : weeklyRecurringNames.join('・'); }
