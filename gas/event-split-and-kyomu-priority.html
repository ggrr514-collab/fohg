} else if (closes.indexOf(ch) >= 0 && depth > 0) { depth--; } if (depth === 0 && (ch === '\n' || ch === '　' || ch === ' ')) { items.push(cur); cur = ''; continue; } cur += ch; } items.push(cur); return items.map(s => s.trim()).filter(Boolean);
